cmake_minimum_required(VERSION 3.10)
project(hgs CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_executable(hgs main.cpp instance.cpp hgs.cpp)
if(MSVC)
  target_compile_options(hgs PRIVATE /O2 /utf-8 /W3 /EHsc)
  target_compile_definitions(hgs PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
else()
  target_compile_options(hgs PRIVATE -O2 -Wall -Wextra)
endif()
